การทำงานของระบบฐานข้อมูล:

หากยังไม่ใส่ URL ระบบจะทำงานใน โหมดทดสอบ (Local Storage) ซึ่งบันทึกข้อมูลและทดสอบหยิบ/แพ็คสินค้าได้ทันที และเมื่อคุณนำ Web App URL จาก Google Apps Script มาวางด้านล่าง ระบบจะบันทึกและซิงค์ออเดอร์ลง Google Sheets ของคุณโดยอัตโนมัติ

สถานะปัจจุบัน: ใช้ข้อมูลจำลองในเครื่อง (Local Storage)
โค้ด Google Apps Script (คัดลอกไปวางใน Extensions > Apps Script) 
function doGet(e) {
  var sheet = getOrCreateSheet();
  var data = sheet.getDataRange().getValues();
  var orders = [];
  for (var i = 1; i < data.length; i++) {
    if (!data[i][0]) continue;
    orders.push({
      orderId: data[i][0],
      orderDate: data[i][1],
      customerName: data[i][2],
      phone: String(data[i][3]),
      address: data[i][4],
      paymentMethod: data[i][5],
      note: data[i][6],
      items: JSON.parse(data[i][7] || "[]"),
      subtotal: Number(data[i][8]),
      shipping: Number(data[i][9]),
      total: Number(data[i][10]),
      status: data[i][11] || "Pending"
    });
  }
  return ContentService.createTextOutput(JSON.stringify({ status: "success", orders: orders.reverse() }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var sheet = getOrCreateSheet();
  var payload = JSON.parse(e.postData.contents);
  
  if (payload.action === "createOrder") {
    var o = payload.order;
    sheet.appendRow([
      o.orderId, o.orderDate, o.customerName, "'" + o.phone,
      o.address, o.paymentMethod, o.note, JSON.stringify(o.items),
      o.subtotal, o.shipping, o.total, o.status
    ]);
    return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  
  if (payload.action === "updateStatus") {
    var data = sheet.getDataRange().getValues();
    for (var i = 1; i < data.length; i++) {
      if (data[i][0] === payload.orderId) {
        sheet.getRange(i + 1, 12).setValue(payload.status);
        break;
      }
    }
    return ContentService.createTextOutput(JSON.stringify({ status: "updated" }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function getOrCreateSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Orders");
  if (!sheet) {
    sheet = ss.insertSheet("Orders");
    sheet.appendRow(["OrderID", "OrderDate", "CustomerName", "Phone", "Address", "PaymentMethod", "Note", "ItemsJSON", "Subtotal", "Shipping", "Total", "Status"]);
  }
  return sheet;
}
